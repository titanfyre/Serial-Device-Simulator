cmake_minimum_required(VERSION 3.20)
cmake_policy(SET CMP0091 NEW)
project(SerialDeviceSimulator VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Use Visual Studio 2022 C++ tools for Windows x64.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(SDK_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/SDK" CACHE PATH "Matching Vdevice SDK")
option(ENABLE_ASAN "Instrument executable code with AddressSanitizer" OFF)
add_compile_definitions(UNICODE _UNICODE NOMINMAX)
add_compile_options(/W4 /WX /utf-8 /permissive- /EHsc /guard:cf)
add_link_options(/guard:cf /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA)
if(ENABLE_ASAN)
  add_compile_options(/fsanitize=address /Zi)
  add_link_options(/INCREMENTAL:NO /DEBUG)
endif()
add_library(SimulatorModel App/Protocol.cpp)
target_include_directories(SimulatorModel PUBLIC App)
add_library(SimulatorSession App/Session.cpp App/Cdc.cpp "${SDK_DIRECTORY}/Vdevice.cpp")
target_include_directories(SimulatorSession PUBLIC "${SDK_DIRECTORY}" App)
target_link_libraries(SimulatorSession PUBLIC SimulatorModel setupapi cfgmgr32 ole32 advapi32)
add_executable(SerialDeviceSimulator WIN32 App/Main.cpp)
target_link_libraries(SerialDeviceSimulator PRIVATE SimulatorSession comctl32 comdlg32 shell32 user32 gdi32 gdiplus dwmapi)
target_link_options(SerialDeviceSimulator PRIVATE "/MANIFESTUAC:level='asInvoker' uiAccess='false'")
add_executable(SerialDeviceSimulatorCLI App/Console.cpp)
target_link_libraries(SerialDeviceSimulatorCLI PRIVATE SimulatorSession)
add_executable(ProtocolTests Tests/ProtocolTests.cpp)
target_link_libraries(ProtocolTests PRIVATE SimulatorModel)
add_executable(ReadinessTests Tests/ReadinessTests.cpp)
target_include_directories(ReadinessTests PRIVATE App)
enable_testing()
add_test(NAME ProtocolModel COMMAND ProtocolTests)
add_test(NAME ReadinessPolicy COMMAND ReadinessTests)
add_test(NAME OfflineCli COMMAND SerialDeviceSimulatorCLI --self-test)
set_target_properties(SerialDeviceSimulator SerialDeviceSimulatorCLI ProtocolTests ReadinessTests PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(config Debug Release RelWithDebInfo MinSizeRel)
  string(TOUPPER "${config}" upper)
  set_target_properties(SerialDeviceSimulator SerialDeviceSimulatorCLI ProtocolTests ReadinessTests PROPERTIES "RUNTIME_OUTPUT_DIRECTORY_${upper}" "${CMAKE_BINARY_DIR}/bin")
endforeach()
